New contract (write exactly; consumed by 02-05/02-06/02-07):

```ts
export interface ProviderLimits { minFrames: number; maxFrames: number }
export interface ProviderQuote { credits: number; available: number }
export interface ProviderStatus {
  status: string;                       // PENDING | PROCESSING | COMPLETED | FAILED | CANCELLED | unknown (open enum)
  progress: number | null;              // never treat null as 0
  error?: { code: string; message: string };
  issues?: { code: string; message: string }[];
}
export type ExportKind = 'mesh-simplified' | 'floorplan';
export type ArtifactKind = 'splat' | 'collider' | 'floorplan' | 'floorplan_meta' | 'cameras';
export interface Artifact { kind: ArtifactKind; name: string; contentType: string; ref: string }
export interface ReconProvider {
  readonly mode: 'mock' | 'live';
  limits(): Promise<ProviderLimits>;                                   // FREE
  quote(frameCount: number): Promise<ProviderQuote>;                   // FREE
  submit(body: string, idempotencyKey: string): Promise<{ requestId: string }>; // SPENDS CREDITS
  status(requestId: string): Promise<ProviderStatus>;                  // FREE
  ensureExport(requestId: string, kind: ExportKind): Promise<'pending' | 'ready' | 'failed'>; // FREE
  artifacts(requestId: string): Promise<Artifact[]>;                   // FREE
  open(artifact: Artifact): Promise<Readable>;                         // FREE (download)
  cancel(requestId: string): Promise<void>;
  verifyWebhook(raw: Buffer, signatureHeader: string | undefined): boolean;
}
// ProviderError gains: code?: string; retryAfterS?: number (constructor opts extended)
export function signWebhook(secret: string, raw: Buffer): string;          // 'sha256=' + hex HMAC-SHA256
export function verifyWebhookSignature(secret: string | undefined, raw: Buffer, header: string | undefined): boolean; // secretEquals, false when secret unset
```
